```ts
// src/app/api/chat/route.ts
import { NextResponse } from 'next/server';
import { OpenAIService } from '../../../lib/services/openai';
import { getServerSession } from 'next-auth/next';
import { authConfig } from '../../../lib/auth/config';
import { prisma } from '../../../lib/db';
import { truncateFilingContent, getTokenCount } from '../../../lib/utils/filing-truncator';
import { encode } from 'gpt-tokenizer';

const MAX_TOTAL_TOKENS = 120000; // Leave buffer for response
const MAX_TOKENS_PER_FILING = 35000; // Rough budget per filing
const SYSTEM_PROMPT_TOKENS = 500; // Approximate tokens for system prompt
const MESSAGE_HISTORY_BUDGET = 4000; // Reserve tokens for message history

interface Filing {
  content: string;
  form: string;
  filingDate: string;
  accessionNumber: string;
  companyName: string;
  textUrl?: string;
  symbol?: string;
  cik?: string;
  htmlUrl?: string;
}

export async function POST(request: Request) {
  try {
    const session = await getServerSession(authConfig);
    if (!session?.user?.id) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }

    // Credit check...
    const user = await prisma.user.findUnique({
      where: { id: session.user.id },
      select: {
        credits: true,
        paidCredits: true
      }
    });

    if (!user || (user.credits === 0 && user.paidCredits === 0)) {
      return NextResponse.json(
        { error: 'No credits remaining.' },
        { status: 403 }
      );
    }

    const body = await request.json();
    const { messages, filings, chatId } = body;

    if (!filings?.length) {
      return NextResponse.json(
        { error: 'At least one filing is required' },
        { status: 400 }
      );
    }

    // Calculate tokens used by message history
    const messageTokens = messages.reduce((sum, msg) => 
      sum + encode(msg.content).length, 0);

    // Calculate available tokens for filings
    const availableFilingTokens = MAX_TOTAL_TOKENS - SYSTEM_PROMPT_TOKENS - messageTokens - MESSAGE_HISTORY_BUDGET;
    const tokensPerFiling = Math.min(
      MAX_TOKENS_PER_FILING,
      Math.floor(availableFilingTokens / filings.length)
    );

    console.log(`Token budget per filing: ${tokensPerFiling}`);

    // Process filings with individual token budgets
    const processedFilings = await Promise.all(filings.map(async (filing: Filing) => {
      const cikMatch = filing.textUrl?.match(/data\/(\d{10})/);
      const cik = cikMatch ? cikMatch[1] : '';

      // Process content with token budget
      const processedContent = truncateFilingContent(filing.content, tokensPerFiling);
      
      return {
        ...filing,
        cik,
        content: processedContent
      };
    }));

    // Create system message
    const systemMessage = {
      role: 'system',
      content: `You are a professional financial analyst examining multiple SEC filings from ${processedFilings[0].companyName}.

Available filings:
${processedFilings.map(f => `- ${f.form} from ${f.filingDate}`).join('\n')}

Filings content:
${processedFilings.map(f => `
=== ${f.form} Filing Date: ${f.filingDate} ===
${f.content}
`).join('\n\n')}

When analyzing:
- Compare metrics and information across filings
- Identify trends and pattern changes
- Note significant year-over-year or quarter-over-quarter changes
- Highlight any shifts in business strategy or risk factors
- Address specific aspects mentioned in the user's question`
    };

    // Generate AI response
    const response = await OpenAIService.generateResponse([systemMessage, ...messages]);

    // Update credits and save chat history...
    if (user.paidCredits > 0) {
      await prisma.user.update({
        where: { id: session.user.id },
        data: { paidCredits: user.paidCredits - 1 }
      });
    } else {
      await prisma.user.update({
        where: { id: session.user.id },
        data: { credits: user.credits - 1 }
      });
    }

    // Save chat history with processed filings
    const chat = await prisma.chat.upsert({
      where: { id: chatId || 'new' },
      create: {
        userId: session.user.id,
        filing: {
          filings: processedFilings.map(f => ({
            form: f.form,
            filingDate: f.filingDate,
            accessionNumber: f.accessionNumber,
            companyName: f.companyName,
            content: f.content,
            textUrl: f.textUrl || '',
            symbol: f.symbol || '',
            cik: f.cik || '',
            htmlUrl: f.htmlUrl || ''
          }))
        },
        messages: [
          ...messages,
          {
            role: 'assistant',
            content: response,
            timestamp: new Date().toISOString()
          }
        ]
      },
      update: {
        messages: [
          ...messages,
          {
            role: 'assistant',
            content: response,
            timestamp: new Date().toISOString()
          }
        ],
        filing: {
          filings: processedFilings.map(f => ({
            form: f.form,
            filingDate: f.filingDate,
            accessionNumber: f.accessionNumber,
            companyName: f.companyName,
            content: f.content,
            textUrl: f.textUrl || '',
            symbol: f.symbol || '',
            cik: f.cik || '',
            htmlUrl: f.htmlUrl || ''
          }))
        },
        updatedAt: new Date()
      }
    });
    
    return NextResponse.json({
      message: response,
      chatId: chat.id
    });

  } catch (error: any) {
    console.error('Chat API error:', error);
    return NextResponse.json(
      {
        message: 'I apologize, but I encountered an error processing the filings. Please try asking about fewer filings at once or focus on specific sections.',
        error: error.message
      },
      { status: error.status || 500 }
    );
  }
}
```

```ts
// src/app/api/credits/route.ts
import { NextResponse } from 'next/server';
import { getServerSession } from 'next-auth/next';
import { authConfig } from '../../../lib/auth/config';
import { prisma } from '../../../lib/db';

export async function GET() {
  try {
    const session = await getServerSession(authConfig);
    if (!session?.user?.id) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }

    const user = await prisma.user.findUnique({
      where: { id: session.user.id },
      select: {
        credits: true,
        paidCredits: true,
        lastCreditRefresh: true
      }
    });

    if (!user) {
      return NextResponse.json({ error: 'User not found' }, { status: 404 });
    }

    // Check if we need to refresh daily credits (at UTC midnight)
    const now = new Date();
    const lastRefresh = new Date(user.lastCreditRefresh || 0);
    
    if (now.getUTCDate() !== lastRefresh.getUTCDate() || 
        now.getUTCMonth() !== lastRefresh.getUTCMonth() ||
        now.getUTCFullYear() !== lastRefresh.getUTCFullYear()) {
      // Reset daily credits
      await prisma.user.update({
        where: { id: session.user.id },
        data: {
          credits: 2, // Reset to 2 free daily credits
          lastCreditRefresh: now
        }
      });
      
      return NextResponse.json({
        credits: 2,
        paidCredits: user.paidCredits
      });
    }

    return NextResponse.json({
      credits: user.credits,
      paidCredits: user.paidCredits
    });
  } catch (error) {
    console.error('Credits API error:', error);
    return NextResponse.json({ error: 'Internal server error' }, { status: 500 });
  }
}
```

```ts
// src/app/api/filing-analysis/route.ts
import { NextResponse } from 'next/server';
import { prisma } from '../../../lib/db';
import { OpenAIService } from '../../../lib/services/openai';
import { cleanFilingContent, extractFilingSection } from '../../../lib/utils/sec';
import type { ChatCompletionSystemMessageParam, ChatCompletionUserMessageParam } from 'openai/resources/chat/completions';
import type { FilingAnalysis } from '../../../lib/types/filing';

const defaultAnalysis: FilingAnalysis = {
  revenue: 0,
  netIncome: 0,
  cashFlow: 0,
  segments: [],
  risks: [],
  keyTakeaways: [],
  revenueTrend: '',
  incomeTrend: '',
  cashFlowTrend: ''
};

export async function POST(request: Request) {
  try {
    const body = await request.json();
    
    if (!body?.filing?.content) {
      return NextResponse.json(
        { error: 'Missing filing content' },
        { status: 400 }
      );
    }

    const cleanedContent = cleanFilingContent(body.filing.content);
    const mdaSection = extractFilingSection(cleanedContent, 'mda');
    const riskSection = extractFilingSection(cleanedContent, 'risk_factors');
    const businessSection = extractFilingSection(cleanedContent, 'business');
    const financialSection = extractFilingSection(cleanedContent, 'financial_statements');

    // Financial metrics extraction
    const financialSystemMessage: ChatCompletionSystemMessageParam = {
      role: 'system',
      content: `Extract key financial metrics and return as JSON:
{
  "revenue": number,
  "netIncome": number,
  "cashFlow": number,
  "revenueTrend": "increasing|decreasing|stable",
  "incomeTrend": "increasing|decreasing|stable",
  "cashFlowTrend": "increasing|decreasing|stable"
}`
    };

    const financialPrompt: ChatCompletionUserMessageParam = {
      role: 'user',
      content: `Analyze financial data from:
${mdaSection}
${financialSection}`
    };

    // Segments analysis
    const segmentsSystemMessage: ChatCompletionSystemMessageParam = {
      role: 'system',
      content: `Extract business segments as JSON array:
[{
  "name": string,
  "value": number,
  "percentage": number,
  "description": string
}]`
    };

    const segmentsPrompt: ChatCompletionUserMessageParam = {
      role: 'user',
      content: `Analyze business segments from:
${businessSection}`
    };

    // Risk analysis
    const risksSystemMessage: ChatCompletionSystemMessageParam = {
      role: 'system',
      content: `Extract key risks as JSON array:
[{
  "title": string,
  "description": string,
  "impact": string
}]`
    };

    const risksPrompt: ChatCompletionUserMessageParam = {
      role: 'user',
      content: `Analyze risk factors from:
${riskSection}`
    };

    let analysis = { ...defaultAnalysis };

    // Process financial metrics
    try {
      const financialResponse = await OpenAIService.generateResponse([
        financialSystemMessage,
        financialPrompt
      ]);
      const financialData = JSON.parse(financialResponse);
      
      analysis = {
        ...analysis,
        revenue: financialData.revenue || 0,
        netIncome: financialData.netIncome || 0,
        cashFlow: financialData.cashFlow || 0,
        revenueTrend: financialData.revenueTrend || '',
        incomeTrend: financialData.incomeTrend || '',
        cashFlowTrend: financialData.cashFlowTrend || ''
      };
    } catch (error) {
      console.error('Failed to process financial metrics:', error);
    }

    // Process segments
    try {
      const segmentsResponse = await OpenAIService.generateResponse([
        segmentsSystemMessage,
        segmentsPrompt
      ]);
      const segmentsData = JSON.parse(segmentsResponse);
      
      if (Array.isArray(segmentsData)) {
        analysis.segments = segmentsData.map(segment => ({
          name: segment.name || '',
          value: segment.value || 0,
          percentage: segment.percentage || 0,
          description: segment.description || ''
        }));
      }
    } catch (error) {
      console.error('Failed to process segments:', error);
    }

    // Process risks
    try {
      const risksResponse = await OpenAIService.generateResponse([
        risksSystemMessage,
        risksPrompt
      ]);
      const risksData = JSON.parse(risksResponse);
      
      if (Array.isArray(risksData)) {
        analysis.risks = risksData.map(risk => ({
          title: risk.title || '',
          description: risk.description || '',
          impact: risk.impact || ''
        }));
      }
    } catch (error) {
      console.error('Failed to process risks:', error);
    }

    // Save to database if company ID is provided
    if (body.companyId && body.filing.accessionNumber) {
      try {
        await prisma.filingAnalysis.upsert({
          where: {
            companyId_filingId: {
              companyId: body.companyId,
              filingId: body.filing.accessionNumber
            }
          },
          update: {
            revenue: analysis.revenue,
            netIncome: analysis.netIncome,
            cashFlow: analysis.cashFlow,
            segments: analysis.segments,
            risks: analysis.risks,
            analysis: analysis,
            updatedAt: new Date()
          },
          create: {
            companyId: body.companyId,
            filingId: body.filing.accessionNumber,
            filingType: body.filing.type,
            filingDate: new Date(body.filing.filingDate),
            revenue: analysis.revenue,
            netIncome: analysis.netIncome,
            cashFlow: analysis.cashFlow,
            segments: analysis.segments,
            risks: analysis.risks,
            analysis: analysis
          }
        });
      } catch (dbError) {
        console.error('Database operation failed:', dbError);
        // Continue execution even if database operation fails
      }
    }

    return NextResponse.json(analysis);
  } catch (error) {
    console.error('Analysis error:', error);
    return NextResponse.json(
      { error: 'Failed to analyze filing', details: error instanceof Error ? error.message : 'Unknown error' },
      { status: 500 }
    );
  }
}
```

```ts
  if (isSecGov) {
    await new Promise(resolve => setTimeout(resolve, 100));
  }

  return fetch(url, {
    headers,
    cache: 'no-store'
  });
}

async function getCIK(input: string): Promise<string> {
  // If the input is already a CIK number (10 digits with leading zeros)
  if (/^\d{10}$/.test(input)) {
    return input;
  }

  // If it's a CIK without leading zeros
  if (/^\d+$/.test(input)) {
    return input.padStart(10, '0');
  }

  console.log('Getting CIK for ticker:', input);
  const response = await fetchWithHeaders(`${WWW_SEC_URL}/files/company_tickers.json`);
  
  if (!response.ok) {
    throw new Error(`Failed to fetch CIK data: ${response.status}`);
  }

  const data: CompanyLookupData = await response.json();
  const company = Object.values(data).find(
    (info: CompanyInfo) => info.ticker.toLowerCase() === input.toLowerCase()
  );

  if (!company) {
    throw new Error(`Company not found for ticker: ${input}`);
  }

  return company.cik_str.toString().padStart(10, '0');
}

async function getFilings(cik: string) {
  const url = `${DATA_SEC_URL}/submissions/CIK${cik}.json`;
  console.log('Fetching filings from:', url);
  
  const response = await fetchWithHeaders(url);
  if (!response.ok) {
    throw new Error(`Failed to fetch filings: ${response.status}`);
  }

  const data = await response.json();
  const filings: any[] = [];
  
  const recent = data.filings.recent;
  for (let i = 0; i < recent.accessionNumber.length; i++) {
    if (recent.form[i] === '10-K' || recent.form[i] === '10-Q') {
      const accessionNumber = recent.accessionNumber[i].replace(/-/g, '');
      const primaryDocument = recent.primaryDocument[i];
      
      filings.push({
        accessionNumber: recent.accessionNumber[i],
        form: recent.form[i],
        type: recent.form[i],
        filingDate: recent.filingDate[i],
        reportDate: recent.reportDate[i],
        primaryDocument,
        textUrl: `${WWW_SEC_URL}/Archives/edgar/data/${cik}/${accessionNumber}/${primaryDocument}`,
        htmlUrl: `${WWW_SEC_URL}/ix?doc=/Archives/edgar/data/${cik}/${accessionNumber}/${primaryDocument}`
      });
    }
  }

  return {
    cik,
    name: data.name,
    filings
  };
}

async function getFilingContent(url: string) {
  try {
    console.log('Fetching filing content from:', url);
    const response = await fetchWithHeaders(url, true);
    
    if (!response.ok) {
      throw new Error(`Failed to fetch document: ${response.status}`);
    }
    
    const rawContent = await response.text();
    
    // Get token count of raw content
    const originalTokenCount = getTokenCount(rawContent);

    // Process the content

    }

  }
}
```

```ts
    const body = await request.text();
    const signature = (await headers()).get('stripe-signature')!;

      );

          }
```

```ts
export async function GET(request: Request) {

    const mode = searchParams.get('mode') || 'search'; // 'search' or 'quote'

    if (!symbol) {
      return NextResponse.json(
        { error: 'Symbol parameter is required' },
        { status: 400 }
      );
    }

    if (mode === 'search') {
      // Company search functionality
      const response = await fetch(
        `https://query1.finance.yahoo.com/v1/finance/search?q=${encodeURIComponent(symbol)}&quotesCount=6&newsCount=0&enableFuzzyQuery=false&quotesQueryId=tss_match_phrase_query`,
        {
          headers: {
            'User-Agent': 'Mozilla/5.0',
            'Accept': '*/*'
          }
        }
      );

      if (!response.ok) {
        throw new Error(`Yahoo Finance API error: ${response.status}`);
      }

      const data = await response.json();
      
      // Filter for relevant US equity results
      const quotes = data.quotes?.filter((quote: any) => 
        quote.quoteType === 'EQUITY' && 
        ['NYQ', 'NMS'].includes(quote.exchange)
      ).map((quote: any) => ({
        symbol: quote.symbol,
        name: quote.longname || quote.shortname,
        exchange: quote.exchange === 'NYQ' ? 'NYSE' : 'NASDAQ'
      })) || [];

      return NextResponse.json({ quotes });

    } else {
      // Basic quote data for a specific symbol
      const response = await fetch(
        `https://query1.finance.yahoo.com/v8/finance/chart/${symbol}?interval=1d`,
        {
          headers: {
            'User-Agent': 'Mozilla/5.0',
            'Accept': '*/*'
          }
        }
      );

      if (!response.ok) {
        throw new Error(`Failed to fetch ${symbol}`);
      }

      const data = await response.json();
      const quote = data?.chart?.result?.[0]?.meta;
      
      return NextResponse.json({
        symbol,
        price: Number(quote?.regularMarketPrice || 0).toFixed(2),
        companyName: quote?.instrumentInfo?.shortName || '',
        exchange: quote?.exchangeName || ''
      });
    }

    console.error('Yahoo Finance API error:', error);
    return NextResponse.json(
      { error: error.message || 'Failed to fetch data' },
      { status: 500 }
    );
  }
```

```tsx
interface ChatHistory {
  id: string;
  filing: {
    filings: Filing[];
  };
  updatedAt: string;
}

export function PreviousChats() {
  const { data: chats, isLoading } = useQuery<ChatHistory[]>({
    queryKey: ['chatHistory'],
    queryFn: async () => {
      const response = await fetch('/api/chat/history');
      if (!response.ok) throw new Error('Failed to fetch chat history');
      return response.json();
    },
    refetchInterval: 3000,
  });

  if (isLoading) return null;
  if (!chats?.length) return null;

  const getChatIdentifier = (chat: ChatHistory): string | null => {
    const filings = chat.filing?.filings;
    if (!filings || filings.length === 0) return null;

    const firstFiling = filings[0];
    if (firstFiling.symbol) return firstFiling.symbol;
    if (firstFiling.cik) return firstFiling.cik;

    const tickerMatch = firstFiling.companyName?.match(/\(([^)]+)\)/);
    if (tickerMatch) return tickerMatch[1];

    return null;
  };

  return (
    <div className="w-full max-w-3xl mx-auto">
      <div className="flex items-center justify-between mb-4">
        <div className="flex items-center gap-2">
          <MessageCircle className="w-4 h-4 text-gray-500" />
```

```tsx
import React, { useState, useRef, useEffect } from 'react';

interface Company {
  symbol: string;
  name: string;
  exchange: string;
}

export function CompanySearch() {
  const [query, setQuery] = useState('');
  const [results, setResults] = useState<Company[]>([]);
  const [loading, setLoading] = useState(false);
  const [focused, setFocused] = useState(false);
  const searchRef = useRef<HTMLDivElement>(null);
  const inputRef = useRef<HTMLInputElement>(null);
  const router = useRouter();

  useEffect(() => {
    function handleClickOutside(event: MouseEvent) {
      if (searchRef.current && !searchRef.current.contains(event.target as Node)) {
        setFocused(false);
        setResults([]);
      }
    }
    document.addEventListener('mousedown', handleClickOutside);
    return () => document.removeEventListener('mousedown', handleClickOutside);
  }, []);

  const searchCompanies = async (searchQuery: string) => {
    if (!searchQuery.trim()) {
      setResults([]);
      return;
    }

    setLoading(true);
    try {
      const response = await fetch(`/api/yahoo?symbol=${encodeURIComponent(searchQuery)}&mode=search`);
      if (!response.ok) throw new Error('Search failed');
      const data = await response.json();
      setResults(data.quotes || []);
    } catch (error) {
      console.error('Search error:', error);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    const timeoutId = setTimeout(() => {
      if (query) {
        searchCompanies(query);
      }
    }, 300);

    return () => clearTimeout(timeoutId);
  }, [query]);

  const handleCompanySelect = (symbol: string) => {
    router.push(`/company/${symbol}`);
    setResults([]);
    setQuery('');
    setFocused(false);
  };

  return (
    <div ref={searchRef} className="relative w-full max-w-3xl mx-auto">
```

```ts
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';

export function useCredits() {
  const queryClient = useQueryClient();

  const { data: credits } = useQuery({
    queryKey: ['credits'],
    queryFn: async () => {
      const response = await fetch('/api/credits');
      if (!response.ok) throw new Error('Failed to fetch credits');
      return response.json();
    }
  });

  const useCredit = useMutation({
    mutationFn: async () => {
      const response = await fetch('/api/credits', {
        method: 'POST'
      });
      if (!response.ok) throw new Error('Failed to use credit');
      return response.json();
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['credits'] });
    }
  });

  return {
    credits: credits?.credits || 0,
    paidCredits: credits?.paidCredits || 0,
    useCredit,
    totalCredits: (credits?.credits || 0) + (credits?.paidCredits || 0)
  };
}
```

```ts
import OpenAI from 'openai';
import type { ChatCompletionMessageParam } from 'openai/resources/chat/completions';

export class OpenAIService {
  private static instance: OpenAI;

  private static getInstance(): OpenAI {
    if (!this.instance) {
      const apiKey = process.env.OPENAI_API_KEY;
      if (!apiKey) {
        throw new Error('OpenAI API key not configured');
      }
      this.instance = new OpenAI({ apiKey });
    }
    return this.instance;
  }

  static async generateResponse(messages: Array<ChatCompletionMessageParam>): Promise<string> {
    try {
      const openai = this.getInstance();

      // Pass messages directly without any truncation
      const response = await openai.chat.completions.create({
        model: 'gpt-4o',
        messages: messages,
        temperature: 0.2,
        presence_penalty: 0.1,
        frequency_penalty: 0.1
      });

      return response.choices[0]?.message?.content || 'I apologize, but I was unable to analyze the filing.';

    } catch (error: any) {
      console.error('OpenAI error:', error);
      if (error.status === 429) {
        return 'I apologize, but I\'m receiving too many requests at the moment. Please try again shortly.';
      }
      return 'I encountered an error analyzing the filing. Please try a more specific question.';
    }
  }
}
```

```ts
// src/lib/utils/filing-truncator.ts
import { encode } from 'gpt-tokenizer';

// Core sections to preserve in order of importance
const CORE_SECTIONS = [
  'Item 7. Management\'s Discussion',
  'Item 2. Management\'s Discussion', // For 10-Q
  'Item 1A. Risk Factors',
  'Item 1. Business',
  'Item 7A. Quantitative and Qualitative Disclosures',
  'Item 3. Quantitative and Qualitative Disclosures' // For 10-Q
];

// Sections to remove
const AUXILIARY_SECTIONS = [
  'Item 15. Exhibits',
  'Item 6. Exhibits',
  'Notes to Consolidated Financial Statements',
  'Signatures',
  'PART III',
  'PART IV',
  'Index to Exhibits',
  'Exhibit Index'
];

function cleanText(text: string): string {
  return text
    .replace(/<[^>]+>/g, ' ') // Remove HTML tags
    .replace(/\s+/g, ' ') // Normalize whitespace
    .replace(/[^\x20-\x7E]/g, '') // Remove non-ASCII characters
    .replace(/Table of Contents/gi, '') // Remove common headers
    .trim();
}

function extractSection(content: string, startMarker: string, endMarkers: string[]): string {
  const startIndex = content.toLowerCase().indexOf(startMarker.toLowerCase());
  if (startIndex === -1) return '';

  let endIndex = content.length;
  for (const marker of endMarkers) {
    const index = content.toLowerCase().indexOf(marker.toLowerCase(), startIndex + startMarker.length);
    if (index !== -1 && index < endIndex) {
      endIndex = index;
    }
  }

  return content.slice(startIndex, endIndex).trim();
}

function splitIntoChunks(text: string, maxTokens: number): string[] {
  const chunks: string[] = [];
  const words = text.split(' ');
  let currentChunk = '';
  let currentTokens = 0;

  for (const word of words) {
    const wordTokens = encode(word + ' ').length;
    if (currentTokens + wordTokens > maxTokens) {
      if (currentChunk) {
        chunks.push(currentChunk.trim());
        currentChunk = '';
        currentTokens = 0;
      }
    }
    currentChunk += word + ' ';
    currentTokens += wordTokens;
  }

  if (currentChunk) {
    chunks.push(currentChunk.trim());
  }

  return chunks;
}

export function truncateFilingContent(content: string, maxTokens: number = 115000): string {
  console.log(`Original content length: ${content.length} chars`);
  
  // First, clean the text
  let processedContent = cleanText(content);
  const originalTokenCount = encode(processedContent).length;
  console.log(`Original token count: ${originalTokenCount}`);

  // If content is already under limit, return cleaned version
  if (originalTokenCount <= maxTokens) {
    console.log(`Content within token limit, returning cleaned version`);
    return processedContent;
  }

  // Extract core sections
  const extractedSections: { content: string; priority: number }[] = [];
  
  // Try to find and extract each core section
  CORE_SECTIONS.forEach((section, index) => {
    const sectionContent = extractSection(
      processedContent,
      section,
      [...CORE_SECTIONS, ...AUXILIARY_SECTIONS].filter(s => s !== section)
    );
    
    if (sectionContent) {
      extractedSections.push({
        content: sectionContent,
        priority: CORE_SECTIONS.length - index
      });
    }
  });

  // If no sections found, fall back to whole document chunks
  if (extractedSections.length === 0) {
    console.log('No sections found, using whole document');
    const chunks = splitIntoChunks(processedContent, maxTokens);
    const result = chunks[0] || ''; // Take first chunk
    console.log(`Final token count (chunked): ${encode(result).length}`);
    return result;
  }

  // Sort sections by priority
  extractedSections.sort((a, b) => b.priority - a.priority);

  // Build final content within token limit
  let finalContent = '';
  let currentTokens = 0;

  for (const section of extractedSections) {
    const sectionTokens = encode(section.content).length;
    
    if (currentTokens + sectionTokens <= maxTokens) {
      // Can add entire section
      finalContent += section.content + '\n\n';
      currentTokens += sectionTokens + 2;
    } else {
      // Add as much of the section as possible
      const remaining = maxTokens - currentTokens;
      if (remaining > 1000) { // Only add partial if we have meaningful space
        const chunks = splitIntoChunks(section.content, remaining);
        if (chunks.length > 0) {
          finalContent += chunks[0] + '\n\n';
          currentTokens += encode(chunks[0]).length + 2;
        }
      }
      break;
    }
  }

  // If we still have no content (shouldn't happen), take the first chunk of the original
  if (!finalContent) {
    console.log('Falling back to first chunk of content');
    const chunks = splitIntoChunks(processedContent, maxTokens);
    finalContent = chunks[0] || '';
  }

  const finalTokenCount = encode(finalContent).length;
  console.log(`Final token count: ${finalTokenCount}`);

  return finalContent.trim();
}

// Helper function to get token count
export function getTokenCount(text: string): number {
  const cleanedText = cleanText(text);
  return encode(cleanedText).length;
}
```

```ts
export function cleanFilingContent(content: string): string {
    return content
      // Remove HTML tags
      .replace(/<[^>]+>/g, ' ')
      // Remove excess whitespace
      .replace(/\s+/g, ' ')
      // Remove special characters
      .replace(/[^\x20-\x7E]/g, '')
      // Remove common SEC headers/footers
      .replace(/Table of Contents/gi, '')
      .replace(/Index to Financial Statements/gi, '')
      .trim();
  }
  
  type FilingSectionMap = {
    [key: string]: [string, string];
  };
  
  export function extractFilingSection(content: string, section: string): string {
    const sectionMarkers: FilingSectionMap = {
      'risk_factors': [
        'Item 1A. Risk Factors',
        'Item 2.'
      ],
      'business': [
        'Item 1. Business',
        'Item 1A.'
      ],
      'mda': [
        'Item 7. Management\'s Discussion and Analysis',
        'Item 8.'
      ],
      'financial_statements': [
        'Item 8. Financial Statements',
        'Item 9.'
      ]
    };
  
    if (!sectionMarkers[section]) {
      return '';
    }
  
    const [startMarker, endMarker] = sectionMarkers[section];
    const startIndex = content.indexOf(startMarker);
    if (startIndex === -1) return '';
  
    const endIndex = content.indexOf(endMarker, startIndex);
    if (endIndex === -1) return content.slice(startIndex);
  
    return content.slice(startIndex, endIndex).trim();
  }
  
  interface FinancialMetrics {
    revenue: number | null;
    netIncome: number | null;
    operatingIncome: number | null;
    assets: number | null;
    liabilities: number | null;
    equity: number | null;
  }
  
  export function summarizeFinancials(content: string): FinancialMetrics {
    // Extract and structure key financial data
    const metrics: FinancialMetrics = {
      revenue: extractMetric(content, 'revenue', 'sales'),
      netIncome: extractMetric(content, 'net income', 'earnings'),
      operatingIncome: extractMetric(content, 'operating income', 'operating profit'),
      assets: extractMetric(content, 'total assets'),
      liabilities: extractMetric(content, 'total liabilities'),
      equity: extractMetric(content, "stockholders' equity", "shareholders' equity")
    };
  
    return metrics;
  }
  
  function extractMetric(content: string, ...terms: string[]): number | null {
    for (const term of terms) {
      const regex = new RegExp(`${term}[^\\d]*(\\d[\\d,]*\\.?\\d*)`, 'i');
      const match = content.match(regex);
      if (match) {
        return parseFloat(match[1].replace(/,/g, ''));
      }
    }
    return null;
  }
  
  export function identifyRisks(content: string): string[] {
    const riskSection = extractFilingSection(content, 'risk_factors');
    const risks: string[] = [];
  
    // Split into paragraphs and identify risk statements
    const paragraphs = riskSection.split(/\n\n+/);
    for (const paragraph of paragraphs) {
      if (paragraph.toLowerCase().includes('risk') || 
          paragraph.toLowerCase().includes('could') ||
          paragraph.toLowerCase().includes('may') ||
          paragraph.toLowerCase().includes('adverse')) {
        risks.push(paragraph.trim());
      }
    }
  
    return risks;
  }
  
  export function extractForwardLooking(content: string): string[] {
    const statements: string[] = [];
    const mdaSection = extractFilingSection(content, 'mda');
  
    // Common forward-looking phrases
    const phrases = [
      'expect',
      'anticipate',
      'believe',
      'plan',
      'intend',
      'estimate',
      'project',
      'forecast'
    ];
  
    // Split into sentences and find forward-looking statements
    const sentences = mdaSection.split(/[.!?]+/);
    for (const sentence of sentences) {
      if (phrases.some(phrase => sentence.toLowerCase().includes(phrase))) {
        statements.push(sentence.trim());
      }
    }
  
    return statements;
  }
```
